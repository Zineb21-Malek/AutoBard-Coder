\section{Fonction d'une variable réelle}

\begin{frame}[shrink=8]{Cadre de travail}
	Dans toute la suite, on considère $E$ et $F$ deux sous-ensembles de $\R$
	(ce que l'on note respectivement $E\subset\R$ et $F\subset\R$).
	\vspace{0.35cm}
	\begin{definitionbox}[Rappel]
		Un sous-ensemble de $\R$ peut être :
		\begin{itemize}
			\item un intervalle : $[a,b]$, $]a,b[$, $[a,+\infty[$, $\R$ ;
			\item une réunion : $\R\setminus\{0\}=]-\infty,0[\,\cup\,]0,+\infty[$ ;
			\item un ensemble fini : $\{0,1,2\}$.
		\end{itemize}
	\end{definitionbox}
	\vspace{0.2cm}
	\begin{applicationbox}
		\faLightbulb\ En gestion, $E$ est souvent $[0,+\infty[$ : une quantité, un prix ou un temps
		ne peut pas être négatif.
	\end{applicationbox}
\end{frame}

\subsection{Définitions}
% ============================================================

\begin{frame}{Qu'est-ce qu'une fonction ?}
	\begin{definitionbox}
		Une \key{fonction d'une variable réelle} c'est la donnée de \important{trois} choses :
		\begin{enumerate}
			\item un ensemble de départ $E$ ;
			\item un ensemble d'arrivée $F$ ;
			\item un procédé qui transforme \important{tout} élément de $E$ en \important{un} élément de $F$.
		\end{enumerate}
	\end{definitionbox}
	\vspace{0.25cm}
	\begin{attentionbox}
		Dans toute la suite on écrira « fonction » plutôt que « fonction d'une variable réelle »,
		par souci de concision.
	\end{attentionbox}
\end{frame}

\begin{frame}{Notation}
	\begin{definitionbox}[Notation]
		Une fonction $f$ sera notée
		\[
			f \colon E \longrightarrow F, \qquad
			x \longmapsto f(x),
		\]
		où, pour tout $x\in E$, $f(x)$ est l'\key{image} de $x$ par la fonction $f$.
	\end{definitionbox}
	\vspace{0.2cm}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{exemplebox}
				$\fonc{f}{\R}{\R}$, $x\longmapsto x^2+1$.\\
				$f(3)=10$ : $10$ est l'image de $3$.
			\end{exemplebox}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{exemplebox}[En gestion]
				$\fonc{C}{[0,+\infty[}{\R_+}$, $q\longmapsto 500+12q$ :\\
				coût de production de $q$ unités.
			\end{exemplebox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Test de la droite verticale}
	\begin{columns}[c]
		\begin{column}{0.52\textwidth}
			Une courbe du plan est le graphe d'une fonction $\R\to\R$
			si et seulement si \important{toute droite verticale} coupe la courbe
			en \important{au plus un} point.
			\vspace{0.3cm}
			\begin{attentionbox}
				Si une même abscisse $x$ correspond à \emph{deux} ordonnées,

\begin{frame}{Domaine de définition}
	\begin{definitionbox}
		Si $\fonc{f}{E}{F}$ est une fonction, l'ensemble $E$ est appelé
		\key{domaine de définition} de $f$. On le note aussi $D_f$.
		En particulier, le domaine de définition ne contient que des éléments
		qui possèdent une image par le procédé $x\longmapsto f(x)$.
	\end{definitionbox}
	\vspace{0.2cm}
	\begin{attentionbox}
		En pratique, si $f$ n'est donnée que par la formule $f(x)$ et que l'on souhaite
		déterminer son domaine \key{maximal} $D_f$, on cherche :
		\begin{itemize}
			\item les $x\in\R$ pour lesquels $f(x)$ \important{n'existe pas}, et on les
			« enlève » de $\R$ (dénominateur qui s'annule) ;
			\item \emph{ou} les $x$ pour lesquels $f(x)$ \important{existe}
			(racine, logarithme).
		\end{itemize}
	\end{attentionbox}
\end{frame}

\begin{frame}{Conditions usuelles sur le domaine}
	\begin{center}
		\renewcommand{\arraystretch}{1.35}
		\begin{tabular}{lll}
			\toprule
			\textbf{Expression} & \textbf{Condition} & \textbf{Exemple} \\
			\midrule
			$\dfrac{1}{g(x)}$ & $g(x)\neq 0$ & $\dfrac{1}{x}$ : $\R\setminus\{0\}$ \\
			$\sqrt{g(x)}$ & $g(x)\ge 0$ & $\sqrt{x+1}$ : $[-1,+\infty[$ \\
			$\dfrac{1}{\sqrt{g(x)}}$ & $g(x)>0$ & $\dfrac{1}{\sqrt{x-2}}$ : $]2,+\infty[$ \\
			$\ln g(x)$ & $g(x)>0$ & $\ln x$ : $]0,+\infty[$ \\
			polynôme & aucune & $x^2-4x+3$ : $\R$ \\
			\bottomrule
		\end{tabular}
	\end{center}
	\vspace{0.15cm}
	\begin{applicationbox}
		On combine les conditions avec des \key{intersections} d'ensembles.
	\end{applicationbox}
\end{frame}

\begin{frame}{Exemples --- domaines maximaux}
	Déterminer les domaines maximaux $E_1$, $E_2$ et $E_3$ des fonctions
	\[
		f_1(x)=\frac{1}{x},\qquad
		f_2(x)=\sqrt{x+1},\qquad
		f_3(x)=\frac{1}{\sqrt{x-2}}.
	\]
	\vspace{0.1cm}
	\begin{exemplebox}[Correction]
		\begin{itemize}
			\item $\frac{1}{x}$ n'est pas défini seulement lorsque $x=0$, ainsi
			$E_1=\R\setminus\{0\}$.
			\item $\sqrt{x+1}$ est défini lorsque $x+1\ge 0$, c'est-à-dire $x\ge -1$.
			Donc $E_2=[-1,+\infty[$.
			\item $\frac{1}{\sqrt{x-2}}$ est défini lorsque $x-2\ge 0$ \emph{et}
			$\sqrt{x-2}\neq 0$, i.e. $x\ge 2$ et $x\neq 2$.
			On en déduit $E_3=]2,+\infty[$.
		\end{itemize}
	\end{exemplebox}
\end{frame}

\begin{frame}{Égalité de deux fonctions}
	\begin{definitionbox}[Proposition 2.1]
		Deux fonctions $\fonc{f_1}{E_1}{F_1}$ et $\fonc{f_2}{E_2}{F_2}$ sont
		\key{égales} si et seulement si les trois points suivants sont vérifiés :
		\begin{enumerate}
			\item $E_1=E_2$ (égalité des ensembles de départ) ;
			\item $F_1=F_2$ (égalité des ensembles d'arrivée) ;
			\item $\forall x\in E_1=E_2$, $f_1(x)=f_2(x)$ (égalité du procédé).
		\end{enumerate}
		Si ces trois propriétés sont vérifiées, on note alors $f_1=f_2$.
	\end{definitionbox}
	\vspace{0.15cm}
	\begin{exemplebox}
		Les fonctions
		$\fonc{f}{\R}{\R}$, $x\longmapsto x+1$
		et
		$\fonc{g}{[0,1]}{\R}$, $x\longmapsto x+1$
		\important{ne sont pas égales} : les ensembles de départ ne sont pas les mêmes.
	\end{exemplebox}
\end{frame}

\begin{frame}{Exercice d'application --- définitions}
	\begin{exercicebox}[Exercice 2.1]
		\begin{enumerate}
			\item Déterminer les domaines maximaux de
			\[
				u(x)=\frac{1}{x^2-9},\quad
				v(x)=\sqrt{4-x},\quad
				w(x)=\frac{1}{\sqrt{x+2}}.
			\]
			\item Un coût de production est donné par
			$C(q)=200+15q+\dfrac{40}{q-8}$ pour $q\ge 0$.
			Quel est le domaine \emph{mathématique} de $C$ ? Quel sous-ensemble
			a un sens \emph{en gestion} ?
			\item Les fonctions $\fonc{f}{\R_+}{\R}$, $x\longmapsto \sqrt{x}$ et
			$\fonc{g}{\R_+}{\R_+}$, $x\longmapsto \sqrt{x}$ sont-elles égales ? Pourquoi ?
		\end{enumerate}
	\end{exercicebox}
\end{frame}

\begin{frame}{Correction --- exercice 2.1}
	\begin{definitionbox}[Correction]
		\begin{enumerate}
			\item $x^2-9\neq 0\iff x\neq\pm 3$, donc $D_u=\R\setminus\{-3,3\}$.\\
			$4-x\ge 0\iff x\le 4$, donc $D_v=]-\infty,4]$.\\
			$x+2>0\iff x>-2$, donc $D_w=]-2,+\infty[$.
			\item Mathématiquement, $q\neq 8$ et $q\in\R$. En gestion : quantité
			positive, et le modèle « explose » en $q=8$, donc
			$D_C^{\text{gest}}=[0,8[\,\cup\,]8,+\infty[$. On travaillera souvent sur $[0,8[$.
			\item \important{Non} : les ensembles d'arrivée $\R$ et $\R_+$ sont distincts
			(proposition 2.1, point 2).
		\end{enumerate}
	\end{definitionbox}
\end{frame}

\subsection{Monotonie}
% ============================================================

\begin{frame}{Croissance et décroissance}
	\begin{definitionbox}
		Soit $\fonc{f}{E}{\R}$ une fonction définie sur un sous-ensemble $E$ de $\R$.
		On dit que :
		\begin{itemize}
			\item $f$ est \key{croissante} sur $E$ si
			$\forall x,y\in E$, $x\le y \Rightarrow f(x)\le f(y)$ ;
			\item $f$ est \key{strictement croissante} sur $E$ si
			$\forall x,y\in E$, $x< y \Rightarrow f(x)< f(y)$ ;
			\item $f$ est \key{décroissante} sur $E$ si
			$\forall x,y\in E$, $x\le y \Rightarrow f(x)\ge f(y)$ ;
			\item $f$ est \key{strictement décroissante} sur $E$ si
			$\forall x,y\in E$, $x< y \Rightarrow f(x)> f(y)$.
		\end{itemize}
	\end{definitionbox}
\end{frame}

\begin{frame}{Lecture graphique}
	\begin{columns}[c]
		\begin{column}{0.48\textwidth}
			\centering
			{\scriptsize Fonction \key{strictement croissante}}\\
			\begin{tikzpicture}

\begin{frame}[shrink=12]{À retenir --- monotonie}
	\begin{center}
		\renewcommand{\arraystretch}{1.25}
		\small
		\begin{tabular}{lll}
			\toprule
			\textbf{Type} & \textbf{Inégalité sur $f$} & \textbf{Graphe} \\
			\midrule
			strict.\ croissant & $x<y\Rightarrow f(x)<f(y)$ & « monte » \\
			croissant & $x\le y\Rightarrow f(x)\le f(y)$ & « ne descend pas » \\
			strict.\ décroissant & $x<y\Rightarrow f(x)>f(y)$ & « descend » \\
			décroissant & $x\le y\Rightarrow f(x)\ge f(y)$ & « ne monte pas » \\
			\bottomrule
		\end{tabular}
	\end{center}
	\vspace{0.2cm}
	\begin{applicationbox}
		\faChartLine\ En gestion : une \key{fonction de demande} $D(p)$ est en général
		décroissante. Une \key{fonction de coût} $C(q)$ est en général croissante.
	\end{applicationbox}
	\vspace{0.15cm}
	\begin{attentionbox}
		La monotonie se lit \important{sur un intervalle}. $x\mapsto x^2$ n'est ni croissante
		ni décroissante sur $\R$ : strictement décroissante sur $]-\infty,0]$,
		strictement croissante sur $[0,+\infty[$.
	\end{attentionbox}
\end{frame}

			$h$ n'est pas monotone sur $\R$ : $h(-2)=4>h(0)=0<h(2)=4$.

\subsection{Parité et périodicité}
% ============================================================

\begin{frame}{Fonctions paires et impaires}
	\begin{definitionbox}
		Soient $I$ un intervalle de $\R$ \key{symétrique par rapport à $0$}
		(de la forme $[-a,a]$, $]-a,a[$ ou $\R$) et $\fonc{f}{I}{\R}$.
		\begin{itemize}
			\item $f$ est \key{paire} si $\forall x\in I$, $f(-x)=f(x)$.\\
			Son graphe est alors \important{symétrique par rapport à l'axe des ordonnées}.
			\item $f$ est \key{impaire} si $\forall x\in I$, $f(-x)=-f(x)$.\\
			Son graphe est alors \important{symétrique par rapport à l'origine} $(0,0)$.
		\end{itemize}
	\end{definitionbox}
	\vspace{0.15cm}
	\begin{attentionbox}
		Si $D_f$ n'est pas symétrique par rapport à $0$, la question de la parité
		ne se pose pas.
	\end{attentionbox}
\end{frame}

\begin{frame}{Périodicité}
	\begin{definitionbox}
		Soient $\fonc{f}{\R}{\R}$ une fonction et un réel $T>0$.
		La fonction $f$ est dite \key{périodique de période $T$} si
		\[
			\forall x\in\R,\qquad f(x+T)=f(x).
		\]
		La plus petite telle période (si elle existe) est la \key{période fondamentale}.
	\end{definitionbox}
	\vspace{0.15cm}
	\begin{exemplebox}
		Les fonctions cosinus et sinus sont $2\pi$-périodiques :
		$\cos(x+2\pi)=\cos x$ et $\sin(x+2\pi)=\sin x$ pour tout $x\in\R$.
	\end{exemplebox}
\end{frame}

			$f(x)=x^2$, $g(x)=x^3$, $h(x)=\lvert x\rvert$, $k(x)=x+x^2$,
			$c(x)=\cos x$, $s(x)=\sin x$.
			\item La fonction $u(x)=\cos(2x)$ est-elle périodique ? Si oui, donner une période.
			Même question pour $v(x)=\sin(x/2)$.
			\item Le chiffre d'affaires mensuel (en kDH) d'une boutique est modélisé par
			\[
				\mathrm{CA}(t)=50+10\cos\!\left(\frac{2\pi t}{12}\right),

			$k(-x)=-x+x^2\neq \pm k(x)$ en général : $k$ n'est ni paire ni impaire.

			La période fondamentale est $T=\pi$.\\
			$v(x+T)=\sin(\frac{x+T}{2})=\sin(\frac{x}{2})$ si $\frac{T}{2}=2\pi$, d'où $T=4\pi$.
			\item $\mathrm{CA}(t+12)=\mathrm{CA}(t)$ : période de $12$ mois, soit
			\key{une année} de saisonnalité (pic / creux qui se répètent chaque année).
		\end{enumerate}
	\end{definitionbox}
\end{frame}

\subsection{Composition de deux fonctions}
% ============================================================

\begin{frame}{Définition de la composée}
	\begin{definitionbox}
		Soient $E,F,G,H$ des sous-ensembles de $\R$ et $\fonc{f}{E}{F}$, $\fonc{g}{G}{H}$
		deux fonctions. Si l'ensemble d'arrivée $F$ de $f$ est inclus dans l'ensemble
		de départ $G$ de $g$, alors on définit la fonction \key{composée} $g\circ f$ par
		\[
			g\circ f \colon E \longrightarrow H, \qquad
			x \longmapsto g\bigl(f(x)\bigr).
		\]
	\end{definitionbox}
	\vspace{0.2cm}
	\centering
	\begin{tikzpicture}[
		box/.style={draw=DarkBlue, rounded corners, fill=DarkBlue!6,
			minimum width=1.6cm, minimum height=0.7cm, font=\small},
		arr/.style={-{Latex}, thick, DarkBlue}]
		\node[box] (x) {$x$};
		\node[box, right=1.6cm of x] (fx) {$f(x)$};
		\node[box, right=1.6cm of fx] (gfx) {$g(f(x))$};
		\draw[arr] (x) -- node[above, font=\small] {$f$} (fx);
		\draw[arr] (fx) -- node[above, font=\small] {$g$} (gfx);
		\draw[arr, Mustard] ([yshift=-0.55cm]x.south) -- node[below, font=\small, Mustard] {$g\circ f$} ([yshift=-0.55cm]gfx.south);
	\end{tikzpicture}
\end{frame}

\begin{frame}{Deux précautions}
	\begin{attentionbox}
		\begin{enumerate}
			\item La condition $F\subset G$ est \important{essentielle} pour que $g(f(x))$ ait
			toujours un sens. De même, $H\subset E$ est essentielle pour que $f(g(x))$ ait un sens.
			\item Il faut faire attention à l'\key{ordre} : en général $f\circ g$ et $g\circ f$
			\important{ne sont pas égales}. Il existe même des cas où $f\circ g$ existe
			alors que $g\circ f$ n'existe pas.
		\end{enumerate}
	\end{attentionbox}
	\vspace{0.25cm}
	\begin{applicationbox}
		\faChartLine\ En gestion : $t$ (heures-machine) $\xrightarrow{f}$ quantité produite
		$\xrightarrow{g}$ revenu. Le revenu en fonction du temps est $g\circ f$.
	\end{applicationbox}
\end{frame}

\begin{frame}{Exemple guidé}
	On considère
	\[
		\begin{array}{c@{\qquad}c@{\qquad}c}
			\fonc{f}{\R}{[-1,1]} &
			\fonc{g}{\R_+}{\R_+} &
			\fonc{h}{\R}{\R_+} \\[2pt]
			x\longmapsto \sin x &
			x\longmapsto \sqrt{x} &
			x\longmapsto x^2.
		\end{array}
	\]
	Peut-on définir $f\circ g$, $g\circ f$, $g\circ h$ et $h\circ f$ ? Si oui, les écrire.
\end{frame}

\begin{frame}{Correction de l'exemple}
	\begin{exemplebox}[Correction]
		\begin{enumerate}
			\item $g\circ f$ : $[-1,1]\not\subset \R_+$, donc $g\circ f$ \important{n'a pas de sens}
			(on ne peut pas prendre $\sqrt{\sin x}$ pour tout $x$, p.ex. $x=\frac{3\pi}{2}$).
			\item $f\circ g$ : $\R_+\subset\R$, donc $f\circ g$ a un sens :
			$\fonc{f\circ g}{\R_+}{[-1,1]}$, $x\longmapsto \sin\sqrt{x}$.
			\item $g\circ h$ : $\R_+\subset\R_+$, donc $g\circ h$ a un sens :
			$\fonc{g\circ h}{\R}{\R_+}$, $x\longmapsto \sqrt{x^2}=\lvert x\rvert$.
			\item $h\circ f$ : $[-1,1]\subset\R$, donc $h\circ f$ a un sens :
			$\fonc{h\circ f}{\R}{\R_+}$, $x\longmapsto (\sin x)^2$.
		\end{enumerate}
	\end{exemplebox}
\end{frame}

\begin{frame}{Exercice d'application --- composition}
	\begin{exercicebox}[Exercice 2.4]
		\begin{enumerate}
			\item Soit $f(x)=x^2+1$ et $g(x)=2x-3$, définies de $\R$ dans $\R$.
			Calculer $(g\circ f)(x)$ et $(f\circ g)(x)$. A-t-on $f\circ g=g\circ f$ ?
			\item Une machine produit $q=f(t)=20t$ unités en $t$ heures.
			Chaque unité se vend $12$ DH, d'où $R=g(q)=12q$.
			\begin{enumerate}
				\item Justifier que $g\circ f$ a un sens et l'expliciter.
				\item $f\circ g$ a-t-elle un sens \emph{économique} ?
			\end{enumerate}
			\item Avec $u(x)=\sqrt{x}$ ($D_u=\R_+$) et $v(x)=x-5$ ($D_v=\R$),
			$u\circ v$ et $v\circ u$ existent-elles sur tout le domaine annoncé ?
		\end{enumerate}
	\end{exercicebox}
\end{frame}

\begin{frame}{Correction --- exercice 2.4}
	\begin{definitionbox}[Correction]
		\begin{enumerate}
			\item $(g\circ f)(x)=2(x^2+1)-3=2x^2-1$ ;
			$(f\circ g)(x)=(2x-3)^2+1=4x^2-12x+10$.
			Donc $f\circ g\neq g\circ f$.
			\item $g\circ f(t)=12\cdot 20t=240t$ : revenu (en DH) après $t$ heures.
			$f\circ g$ composerait un \emph{revenu} avec une fonction qui attend un \emph{temps} :
			sans intérêt économique ici.
			\item $v\circ u$ : $\R_+\subset\R$, oui, $(v\circ u)(x)=\sqrt{x}-5$.
			$u\circ v$ : l'arrivée de $v$ est $\R\not\subset\R_+$. On peut toutefois restreindre
			$v$ à $[5,+\infty[$, alors $(u\circ v)(x)=\sqrt{x-5}$.
		\end{enumerate}
	\end{definitionbox}
\end{frame}

\subsection{Antécédent, image directe et image réciproque}
% ============================================================

\begin{frame}{Antécédent et image}
	\begin{definitionbox}
		Soient $\fonc{f}{E}{F}$ une fonction et $B$ un sous-ensemble de $F$.
		\begin{itemize}
			\item On appelle \key{antécédent} de $y\in F$ par $f$ tout $x\in E$ tel que $f(x)=y$.
			On dit alors que $y$ est l'\key{image} de $x$ par $f$.
			\item L'\key{image de $f$} est le sous-ensemble de $F$ noté $I(f)$ défini par
			\[
				I(f)=\bigl\{ y\in F : \exists x\in E,\ y=f(x)\bigr\}.
			\]
			\item L'\key{image réciproque} de $B$ par $f$ est le sous-ensemble de $E$ noté
			$f^{-1}(B)$ défini par
			\[
				f^{-1}(B)=\bigl\{ x\in E : f(x)\in B\bigr\}.
			\]
		\end{itemize}
	\end{definitionbox}
\end{frame}

\begin{frame}{Ne pas confondre $I(f)$ et $f(x)$}
	\begin{attentionbox}
		Il ne faut pas confondre l'image de $f$ notée $I(f)$ et l'image de $x$ par $f$
		notée $f(x)$ : ce ne sont pas le même type d'objet.
		\begin{itemize}
			\item $I(f)$ est un \key{sous-ensemble} de l'espace d'arrivée $F$ ;
			\item $f(x)$ est un \key{élément} de $F$.
		\end{itemize}
		$I(f)$ est l'ensemble des éléments de $F$ qui ont \important{au moins un} antécédent par $f$.
	\end{attentionbox}
	\vspace{0.2cm}
	\begin{exemplebox}
		Pour $f(x)=x^2$ de $\R$ dans $\R$ : $f(2)=4$ (un réel), alors que
		$I(f)=[0,+\infty[$ (un intervalle). Les antécédents de $4$ sont $-2$ et $2$.
	\end{exemplebox}
\end{frame}

\begin{frame}{Image réciproque : lecture graphique}
	\begin{columns}[c]
		\begin{column}{0.50\textwidth}
			Soit $f(x)=x^2$ et $B=[1,4]$. Alors
			\begin{align*}
				f^{-1}(B)
				&=\bigl\{x\in\R : 1\le x^2\le 4\bigr\}\\
				&=[-2,-1]\cup[1,2].
			\end{align*}
			\vspace{0.15cm}
			\begin{applicationbox}
				$f^{-1}(B)$ = les $x$ dont l'image tombe dans $B$.
				Ce n'est \important{pas} la fonction réciproque
				(qui n'existe que si $f$ est bijective).
			\end{applicationbox}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{tikzpicture}
				\begin{axis}[gestion, width=6.4cm, height=5.0cm,
					xmin=-2.6, xmax=2.6, ymin=-0.4, ymax=5.2]
					\addplot[DarkBlue, domain=-2.3:2.3, samples=80] ({x}, {x^2});
					\fill[Mustard, opacity=0.18] (axis cs:-2.6,1) rectangle (axis cs:2.6,4);
					\addplot[only marks, mark=*, DarkRed] coordinates{(-2,4)(-1,1)(1,1)(2,4)};
				\end{axis}
			\end{tikzpicture}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Exercice d'application --- images}
	\begin{exercicebox}[Exercice 2.5]
		Soit $f(x)=x^2-2x$ définie sur $E=\R$.
		\begin{enumerate}
			\item Calculer $f(0)$, $f(1)$, $f(3)$. Quels sont les antécédents de $0$ ?
			\item Déterminer $I(f)$ (on pourra écrire $f(x)=(x-1)^2-1$).
			\item Calculer $f^{-1}\bigl(\{0\}\bigr)$ et $f^{-1}\bigl([0,+\infty[\bigr)$.
			\item \textbf{Gestion.} Le profit (en kDH) est $B(q)=8q-400$, $q$ en unités.
			Interpréter $B^{-1}\bigl([0,+\infty[\bigr)$.
		\end{enumerate}
	\end{exercicebox}
\end{frame}

\begin{frame}{Correction --- exercice 2.5}
	\begin{definitionbox}[Correction]
		\begin{enumerate}
			\item $f(0)=0$, $f(1)=-1$, $f(3)=3$. $f(x)=0\iff x(x-2)=0$, antécédents $0$ et $2$.
			\item $f(x)=(x-1)^2-1\ge -1$, et toutes les valeurs $\ge -1$ sont atteintes :
			$I(f)=[-1,+\infty[$.
			\item $f^{-1}(\{0\})=\{0,2\}$.
			$f(x)\ge 0\iff x(x-2)\ge 0\iff x\in]-\infty,0]\cup[2,+\infty[$.
			\item $B(q)\ge 0\iff q\ge 50$ : $B^{-1}([0,+\infty[)=[50,+\infty[$.
			C'est l'ensemble des productions pour lesquelles l'entreprise
			est \key{au-dessus du seuil de rentabilité}.
		\end{enumerate}
	\end{definitionbox}
\end{frame}

\subsection{Bijectivité et fonction réciproque}
% ============================================================

\begin{frame}{Définition d'une bijection}
	\begin{definitionbox}
		Une fonction $\fonc{f}{E}{F}$ est dite \key{bijective} si tout point $y$ de
		l'espace d'arrivée $F$ possède \important{exactement un} antécédent $x$ dans
		l'espace de départ $E$ par la fonction $f$.
	\end{definitionbox}
	\vspace{0.2cm}
	\begin{attentionbox}
		Avec des quantificateurs :
		\[
			\forall y\in F,\ \exists !\, x\in E \quad\text{tel que}\quad y=f(x).
		\]
		Autrement dit : $f$ est à la fois \key{injective} (au plus un antécédent)
		et \key{surjective} (au moins un antécédent).
	\end{attentionbox}
\end{frame}

\begin{frame}{Test visuel (droite horizontale)}
	\begin{columns}[c]
		\begin{column}{0.50\textwidth}
			$f$ est bijective de $E$ dans $F$ lorsque toute droite horizontale
			$y=y_0$ avec $y_0\in F$ coupe le graphe
			\begin{itemize}
				\item \important{une} fois (existence = surjection) ;
				\item et \important{une seule} fois (unicité = injection).
			\end{itemize}
			\vspace{0.15cm}
			\begin{applicationbox}
				Restreindre le départ et/ou l'arrivée permet souvent
				de « rendre » une fonction bijective.
			\end{applicationbox}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{tikzpicture}
				\begin{axis}[gestion, width=6.3cm, height=5.0cm,
					xmin=-2.3, xmax=2.3, ymin=-0.4, ymax=3.4]
					\addplot[DarkBlue, domain=-1.8:1.8, samples=60] ({x}, {x^2});
					\draw[DarkRed, densely dashed] (axis cs:-2.2,1.2) -- (axis cs:2.2,1.2);

\begin{frame}{Un critère pratique}
	\begin{definitionbox}[Théorème 2.1]
		Si une fonction $\fonc{f}{E}{F}$ est \key{strictement croissante} et telle que
		$F=I(f)$, alors elle est \key{bijective}.
	\end{definitionbox}
	\vspace{0.2cm}
	\begin{attentionbox}
		\begin{itemize}
			\item Le théorème fonctionne aussi si $f$ est \key{strictement décroissante}.
			\item Pour toute $\fonc{f}{E}{F}$, si $F=I(f)$ on dit que $f$ est
			\key{à valeurs dans son image}.
		\end{itemize}
	\end{attentionbox}
	\vspace{0.15cm}
	\begin{exemplebox}
		$\fonc{h}{[0,2]}{[0,4]}$, $x\longmapsto x^2$ est bijective car elle est
		strictement croissante et $I(h)=h([0,2])=[0,4]$.
	\end{exemplebox}
\end{frame}

\begin{frame}{Fonction réciproque}
	\begin{definitionbox}
		Si $\fonc{f}{E}{F}$ est bijective, il existe une fonction appelée
		\key{fonction réciproque} de $f$, notée $f^{-1}$, définie par
		\[
			f^{-1} \colon F \longrightarrow E, \qquad
			y \longmapsto \text{l'unique } x\in E \text{ tel que } y=f(x).
		\]
	\end{definitionbox}
	\vspace{0.15cm}
	\begin{attentionbox}
		La condition « $f$ bijective » est \important{essentielle} pour que le $x$ tel que
		$y=f(x)$ soit bien défini de manière unique.
		Ne pas confondre $f^{-1}$ (la réciproque) et $f^{-1}(B)$ (l'image réciproque d'un ensemble),
		ni avec $1/f$.
	\end{attentionbox}
\end{frame}

\begin{frame}{Exemple --- réciproque de $x^2$ sur $[0,2]$}
	La fonction $\fonc{h}{[0,2]}{[0,4]}$, $x\longmapsto x^2$ est bijective.
	Pour tout $y\in[0,4]$, l'équation $y=x^2$ admet pour unique solution
	$x=\sqrt{y}\in[0,2]$. Donc
	\[
		h^{-1} \colon [0,4]\longrightarrow [0,2], \qquad y\longmapsto \sqrt{y}.
	\]
	\vspace{0.1cm}
	\begin{columns}[c]
		\begin{column}{0.48\textwidth}
			\begin{tikzpicture}
				\begin{axis}[gestion, width=6.2cm, height=4.4cm,
					xmin=-0.3, xmax=4.4, ymin=-0.3, ymax=4.4,
					legend style={at={(0.98,0.22)}, anchor=south east}]
					\addplot[DarkBlue, domain=0:2, samples=40] ({x}, {x^2});
					\addlegendentry{$h$}
					\addplot[Mustard, domain=0:4, samples=40] ({x}, {sqrt(x)});
					\addlegendentry{$h^{-1}$}
					\addplot[gray, dashed, domain=0:4] ({x}, {x});
				\end{axis}
			\end{tikzpicture}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{applicationbox}
				Les graphes de $h$ et $h^{-1}$ sont \key{symétriques} par rapport
				à la première bissectrice $y=x$.
			\end{applicationbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Exemple guidé --- $f(x)=x^2-4x+3$}
	On considère
	\[
		\fonc{f}{]-\infty,2]}{I(f)}, \qquad x\longmapsto x^2-4x+3.
	\]
	Déterminer $I(f)$, montrer que $f$ est une bijection et calculer $f^{-1}$.
	\vspace{0.2cm}
	\begin{exemplebox}[Tableau de variations]
		Pour tout $x\in]-\infty,2]$, $f'(x)=2x-4\le 0$.
		\vspace{0.1cm}
		\centering
		\renewcommand{\arraystretch}{1.25}
		\begin{tabular}{|c|ccc|}
			\hline
			$x$ & $-\infty$ && $2$ \\
			\hline
			$f'(x)$ && $-$ & $0$ \\
			\hline
			$f$ & $+\infty$ & $\searrow$ & $-1$ \\
			\hline
		\end{tabular}
	\end{exemplebox}
\end{frame}

\begin{frame}{Suite de l'exemple --- calcul de $f^{-1}$}
	D'après le tableau, $f$ est strictement décroissante et $I(f)=[-1,+\infty[$ :
	elle est donc bijective de $]-\infty,2]$ dans $I(f)$.
	Soit $y\in[-1,+\infty[$. On a
	\[
		y=f(x)\iff y=x^2-4x+3\iff x^2-4x+(3-y)=0.
	\]
	Le discriminant est $\Delta=16-4(3-y)=4+4y\ge 0$ car $y\ge -1$. Les racines sont
	\[
		x_1=2-\sqrt{1+y},\qquad x_2=2+\sqrt{1+y}.
	\]
	Comme on cherche $x\in]-\infty,2]$, $x_2$ ne convient pas (sauf si $y=-1$).
	\vspace{0.1cm}
	\begin{definitionbox}[Conclusion]
		\[
			\fonc{f^{-1}}{[-1,+\infty[}{]-\infty,2]}, \qquad
			y\longmapsto 2-\sqrt{1+y}.
		\]
	\end{definitionbox}
\end{frame}

\begin{frame}{Composition avec la réciproque}
	\begin{definitionbox}[Proposition 2.2]
		Soient $\fonc{f}{E}{F}$ une fonction bijective et $\fonc{f^{-1}}{F}{E}$ sa réciproque.
		Alors
		\[
			\forall x\in E,\quad f^{-1}\bigl(f(x)\bigr)=x
			\qquad\text{et}\qquad
			\forall y\in F,\quad f\bigl(f^{-1}(y)\bigr)=y.
		\]
	\end{definitionbox}
	\vspace{0.2cm}
	\begin{applicationbox}
		\faLightbulb\ $f$ et $f^{-1}$ « s'annulent » l'une l'autre :
		partir de $x$, appliquer $f$ puis $f^{-1}$, c'est revenir à $x$.
	\end{applicationbox}
\end{frame}

\subsection{Continuité et dérivabilité d'une fonction}
% ============================================================

\begin{frame}{Continuité}
	On note dans la suite $I$ et $J$ des intervalles de $\R$.
	\begin{definitionbox}
		Soient $\fonc{f}{I}{\R}$ et $a\in I$.
		\begin{itemize}
			\item On dit que $f$ est \key{continue au point $a$} si
			$\displaystyle\lim_{x\to a} f(x)=f(a)$.
			\item On dit que $f$ est \key{continue sur $I$} si $f$ est continue en tout point de $I$.
		\end{itemize}
	\end{definitionbox}
	\vspace{0.15cm}
	\begin{applicationbox}
		Graphiquement : on peut tracer le graphe \important{sans lever le crayon} sur $I$.
		Les polynômes, $\exp$, $\ln$ (sur $]0,+\infty[$), $\sin$, $\cos$ sont continues
		sur leur domaine.
	\end{applicationbox}
\end{frame}

\begin{frame}{Dérivabilité}
	\begin{definitionbox}
		Soient $\fonc{f}{I}{\R}$ et $a\in I$.
		\begin{itemize}
			\item $f$ est \key{dérivable en $a$} si
			\[
				x\longmapsto \frac{f(x)-f(a)}{x-a}
			\]
			admet une \important{limite finie} lorsque $x\to a$. Cette limite est notée $f'(a)$
			et appelée \key{dérivée de $f$ en $a$}.
			\item $f$ est dérivable sur $J\subset I$ si elle l'est en tout point de $J$.
			\item Le \key{domaine de dérivabilité} est
			$\widetilde I=\{x\in I : f\text{ est dérivable en }x\}$, et
			$\fonc{f'}{\widetilde I}{\R}$, $x\longmapsto f'(x)$.
		\end{itemize}
	\end{definitionbox}
\end{frame}

\begin{frame}{Exemple --- $f(x)=x^2$}
	La fonction $\fonc{f}{\R}{\R}$, $x\longmapsto x^2$ est dérivable sur $\R$.
	En effet, pour tout $x_0\in\R$,
	\begin{align*}
		\frac{f(x)-f(x_0)}{x-x_0}
		&= \frac{x^2-x_0^2}{x-x_0}
		= \frac{(x-x_0)(x+x_0)}{x-x_0}
		= x+x_0 \\
		&\xrightarrow[x\to x_0]{} 2x_0 < +\infty.
	\end{align*}
	On a même montré que $f'(x_0)=2x_0$, autrement dit
	\[
		f'(x)=2x \qquad \forall x\in\R.
	\]
\end{frame}

\begin{frame}{Dérivées successives}
	\begin{definitionbox}
		Soient $\fonc{f}{I}{\R}$ dérivable et $f'$ sa dérivée. Si $f'$ est aussi dérivable,
		on note $f''=(f')'$ la \key{dérivée seconde} de $f$. Plus généralement,
		\[
			f^{(0)}=f,\quad f^{(1)}=f',\quad f^{(2)}=f'',\quad \ldots,\quad
			f^{(n+1)}=\bigl(f^{(n)}\bigr)'.
		\]
		Si la dérivée $n$-ième $f^{(n)}$ existe, on dit que $f$ est \key{$n$ fois dérivable}.
	\end{definitionbox}
	\vspace{0.2cm}
	\begin{exemplebox}[En gestion]
		$C(q)$ coût total, $C'(q)$ = \key{coût marginal} (coût de la dernière unité),
		$C''(q)$ indique si ce coût marginal augmente ou diminue.
	\end{exemplebox}
\end{frame}

\begin{frame}{Règles de dérivation}
	\begin{definitionbox}[Propriétés]
		\small
		Soient $f,g$ dérivables sur $I$. Pour tout $x\in I$ :
		\begin{itemize}
			\item $(f+g)'(x)=f'(x)+g'(x)$ ;
			\item $(\lambda f)'(x)=\lambda f'(x)$, $\forall\lambda\in\R$ ;
			\item $(f\cdot g)'(x)=f'(x)g(x)+g'(x)f(x)$ ;
			\item $\displaystyle\left(\frac{1}{f}\right)'(x)=-\frac{f'(x)}{f(x)^2}$ si $f(x)\neq 0$ ;
			\item $\displaystyle\left(\frac{f}{g}\right)'(x)=\frac{f'(x)g(x)-f(x)g'(x)}{g(x)^2}$ si $g(x)\neq 0$.
		\end{itemize}
	\end{definitionbox}
\end{frame}

\begin{frame}{Dérivation d'une composée}
	\begin{definitionbox}[Proposition]
		Soient $\fonc{f}{I}{\R}$ et $\fonc{g}{J}{\R}$ dérivables, avec $I(f)\subset J$.
		Alors $g\circ f$ est dérivable sur $I$ et
		\[
			\forall x\in I, \qquad (g\circ f)'(x)=f'(x)\cdot g'\bigl(f(x)\bigr).
		\]
	\end{definitionbox}
	\vspace{0.15cm}
	\begin{exemplebox}
		\begin{enumerate}
			\item $f(x)=e^{x^2+3x+1}=f_1\circ f_2$ avec $f_1(x)=e^x$, $f_2(x)=x^2+3x+1$ :
			$f'(x)=(2x+3)\,e^{x^2+3x+1}$.
			\item $g(x)=\sin(2x+1)$ : $g'(x)=2\cos(2x+1)$.
		\end{enumerate}
	\end{exemplebox}
\end{frame}

\begin{frame}[shrink=8]{Dérivée de la fonction réciproque}
	\begin{definitionbox}[Corollaire]
		Soient $\fonc{f}{I}{J}$ dérivable et bijective, et $f^{-1}$ sa réciproque.
		Si $f'$ ne s'annule pas sur $I$, alors $f^{-1}$ est dérivable et
		\[
			\forall y\in J, \qquad \bigl(f^{-1}\bigr)'(y)=\frac{1}{f'\bigl(f^{-1}(y)\bigr)}.
		\]
	\end{definitionbox}
	\vspace{0.08cm}
	\begin{exemplebox}
		$f(x)=x^2$ sur $[0,2]$, $f^{-1}(y)=\sqrt{y}$. Pour $x\in]0,2[$, $f'(x)=2x\neq 0$.
		Donc sur $]0,4[$,
		\[
			\bigl(f^{-1}\bigr)'(y)=\frac{1}{2f^{-1}(y)}=\frac{1}{2\sqrt{y}},
		\]
		ce qui retrouve la dérivée usuelle de la racine carrée.
	\end{exemplebox}
\end{frame}

\begin{frame}{Tangente en un point}
	\begin{columns}[c]
		\begin{column}{0.50\textwidth}
			\begin{definitionbox}[Tangente]
				Soit $f$ dérivable en $a\in I$. Une équation de la \key{tangente}
				à la courbe $\mathcal{C}_f$ au point $\bigl(a,f(a)\bigr)$ est
				\[
					y=f(a)+f'(a)\,(x-a).
				\]
			\end{definitionbox}
			\vspace{0.15cm}
			\begin{exemplebox}
				Pour $f(x)=x^2$ en $a=1$ : $f(1)=1$, $f'(1)=2$, d'où
				$y=1+2(x-1)=2x-1$.
			\end{exemplebox}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{tikzpicture}
				\begin{axis}[gestion, width=6.3cm, height=5.2cm,
					xmin=-0.4, xmax=2.4, ymin=-1.2, ymax=4.2]
					\addplot[DarkBlue, domain=0:2.1, samples=50] ({x}, {x^2});
					\addplot[Mustard, domain=0:2.2, samples=20] ({x}, {2*x-1});
					\addplot[only marks, mark=*, DarkRed] coordinates{(1,1)};
					\node[font=\scriptsize, DarkRed, left] at (axis cs:1,1.35) {$(1,1)$};
				\end{axis}
			\end{tikzpicture}
		\end{column}
	\end{columns}
\end{frame}

\subsection{Étude des variations d'une fonction}
% ============================================================

\begin{frame}{Sens de variation et signe de $f'$}
	\begin{definitionbox}[Proposition]
		Soient $a,b\in\R$ et $\fonc{f}{[a,b]}{\R}$ continue sur $[a,b]$ et dérivable sur $]a,b[$. Alors :
		\begin{enumerate}
			\item $\forall x\in]a,b[$, $f'(x)\ge 0 \iff f$ est croissante ;
			\item $\forall x\in]a,b[$, $f'(x)\le 0 \iff f$ est décroissante ;
			\item $\forall x\in]a,b[$, $f'(x)=0 \iff f$ est constante ;
			\item $\forall x\in]a,b[$, $f'(x)>0 \Rightarrow f$ est strictement croissante ;
			\item $\forall x\in]a,b[$, $f'(x)<0 \Rightarrow f$ est strictement décroissante.
		\end{enumerate}
	\end{definitionbox}
\end{frame}

\begin{frame}{Une réciproque fausse}
	\begin{attentionbox}
		La réciproque des points 4. et 5. est \important{fausse}.
		Par exemple, $x\longmapsto x^3$ est \key{strictement croissante} sur $\R$,
		et pourtant sa dérivée $3x^2$ \important{s'annule} en $0$.
	\end{attentionbox}
	\vspace{0.15cm}
	\begin{columns}[c]
		\begin{column}{0.48\textwidth}
			\begin{tikzpicture}
				\begin{axis}[gestion, width=6.2cm, height=4.4cm,
					xmin=-1.6, xmax=1.6, ymin=-2.2, ymax=2.2]
					\addplot[DarkBlue, domain=-1.3:1.3, samples=60] ({x}, {x^3});
					\addplot[only marks, mark=*, DarkRed] coordinates{(0,0)};
				\end{axis}
			\end{tikzpicture}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{applicationbox}
				Méthode d'étude :
				\begin{enumerate}
					\item domaine $D_f$ ;
					\item dérivée $f'$ (là où elle existe) ;
					\item signe de $f'$ ;
					\item tableau de variations ;
					\item éventuels extremums.
				\end{enumerate}
			\end{applicationbox}
		\end{column}
	\end{columns}
\end{frame}

\subsection{La valeur absolue}

\begin{frame}[shrink=10]{Définition}
	\begin{columns}[c]
		\begin{column}{0.52\textwidth}
			\begin{definitionbox}
				La fonction \key{valeur absolue} notée $\lvert\,\cdot\,\rvert$ est définie par
				\[
					\begin{array}{ll}
						\fonc{\lvert\,\cdot\,\rvert}{\R}{\R_+} , \\[4pt]
						x\longmapsto
						\begin{cases}
							x & \text{si } x\ge 0, \\
							-x & \text{si } x\le 0.
						\end{cases}
					\end{array}
				\]
			\end{definitionbox}
			\vspace{0.15cm}
			\begin{attentionbox}
				Sur la droite numérique, $\lvert x-y\rvert$ est la \key{distance}
				entre $x$ et $y$. En particulier, $\lvert x\rvert$ est la distance
				de $x$ à $0$.
			\end{attentionbox}
		\end{column}
		\begin{column}{0.46\textwidth}
			\begin{tikzpicture}
				\begin{axis}[gestion, width=5.8cm, height=4.0cm,
					xmin=-2.6, xmax=2.6, ymin=-0.4, ymax=2.6]
					\addplot[DarkBlue, domain=-2.3:0, samples=2] ({x}, {-x});
					\addplot[DarkBlue, domain=0:2.3, samples=2] ({x}, {x});
					\node[font=\small] at (axis cs:1.7,2.15) {$y=\lvert x\rvert$};
				\end{axis}
			\end{tikzpicture}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Propriétés de la valeur absolue (1/2)}
	\begin{definitionbox}[Proposition 3.1]
		Pour tout $x,y\in\R$ :
		\begin{enumerate}
			\item $\lvert x\rvert\ge 0$ ; $\lvert -x\rvert=\lvert x\rvert$ ; et $\lvert x\rvert>0\iff x\neq 0$ ;
			\item $\sqrt{x^2}=\lvert x\rvert$ ;
			\item $\lvert xy\rvert=\lvert x\rvert\cdot\lvert y\rvert$ ;
			\item $\forall r\in\R_+$, $\lvert x\rvert\le r \iff -r\le x\le r \iff x\in[-r,r]$.
		\end{enumerate}
	\end{definitionbox}
\end{frame}

\begin{frame}{Propriétés de la valeur absolue (2/2)}
	\begin{definitionbox}[Proposition 3.1, suite]
		\begin{enumerate}
			\setcounter{enumi}{4}
			\item $\forall b\in\R$,
			$\lvert x\rvert\ge b \iff
			\begin{cases}
				x\ge b\text{ ou }x\le -b & \text{si }b\ge 0,\\
				x\in\R & \text{si }b<0 ;
			\end{cases}$
			\item \key{Inégalité triangulaire} : $\lvert x+y\rvert\le\lvert x\rvert+\lvert y\rvert$ ;
			\item \key{Seconde inégalité triangulaire} :
			$\bigl\lvert \lvert x\rvert-\lvert y\rvert\bigr\rvert\le\lvert x-y\rvert$.
		\end{enumerate}
	\end{definitionbox}
\end{frame}

\begin{frame}{Preuve de l'inégalité triangulaire}
	Soient $x,y$ deux réels quelconques. On a
	\[
		-\lvert x\rvert\le x\le \lvert x\rvert
		\qquad\text{et}\qquad
		-\lvert y\rvert\le y\le \lvert y\rvert.
	\]
	En additionnant ces deux inégalités,
	\[
		-\bigl(\lvert x\rvert+\lvert y\rvert\bigr)
		\le x+y
		\le \lvert x\rvert+\lvert y\rvert.
	\]
	D'après le point 4. de la proposition (avec $r=\lvert x\rvert+\lvert y\rvert$), on obtient
	\[
		\lvert x+y\rvert\le \lvert x\rvert+\lvert y\rvert.
	\]
	\vspace{0.1cm}
	\begin{applicationbox}
		Pour la seconde : $\lvert x\rvert=\lvert (x-y)+y\rvert\le\lvert x-y\rvert+\lvert y\rvert$,
		d'où $\lvert x\rvert-\lvert y\rvert\le\lvert x-y\rvert$. En échangeant $x$ et $y$,
		$\lvert y\rvert-\lvert x\rvert\le\lvert x-y\rvert$, et on conclut.
	\end{applicationbox}
\end{frame}

\begin{frame}{Correction --- exercice 3.1}
	\begin{definitionbox}[Correction]
		\begin{enumerate}
			\item $\lvert 2x-6\rvert<4\iff -4<2x-6<4\iff 1<x<5$, donc $x\in]1,5[$.\\
